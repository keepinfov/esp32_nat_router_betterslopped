if(CONFIG_MQTT_HOMEASSISTANT)
    set(MQTT_HA_SRCS "mqtt_ha.c")
else()
    set(MQTT_HA_SRCS "")
endif()

idf_component_register(
    SRCS ${MQTT_HA_SRCS}
    INCLUDE_DIRS "include"
    PRIV_REQUIRES mqtt esp_event esp_wifi esp_netif esp_timer nvs_flash console cmd_router remote_console
)
# Only when there is something to compile: a component registered with no
# sources is an INTERFACE target, and PRIVATE options on one are a CMake error.
# Without this guard, CONFIG_MQTT_HOMEASSISTANT=n failed to configure at all —
# the option existed but had never been exercised.
if(MQTT_HA_SRCS)
    target_compile_options(${COMPONENT_LIB} PRIVATE -Os)
endif()
